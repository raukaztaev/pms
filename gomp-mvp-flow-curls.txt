# 1) Login organizer
curl -X POST http://localhost:8080/api/v1/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"organizer1@gomp.kz","password":"password"}'

# 2) Login user
curl -X POST http://localhost:8080/api/v1/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"user1@gomp.kz","password":"password"}'

# 3) Create order (replace ORGANIZER_TOKEN)
curl -X POST http://localhost:8080/api/v1/orders \
  -H "Authorization: Bearer ORGANIZER_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "productName":"MVP flow order",
    "sourcePlatform":"Temu",
    "minTargetAmount":1000,
    "deadline":"2026-12-31T10:00:00Z"
  }'

# 4) Join order (replace USER_TOKEN and ORDER_ID)
curl -X POST http://localhost:8080/api/v1/orders/ORDER_ID/join \
  -H "Authorization: Bearer USER_TOKEN" \
  -H "Content-Type: application/json"

# 5) Add item (replace USER_TOKEN and ORDER_ID)
curl -X POST http://localhost:8080/api/v1/orders/ORDER_ID/items \
  -H "Authorization: Bearer USER_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{
    "name":"Phone case",
    "link":"https://temu.com/sample-item",
    "quantity":2,
    "unitPrice":1250,
    "notes":"Black color"
  }'

# 6) Get payments (replace ORGANIZER_TOKEN and ORDER_ID)
curl -X GET http://localhost:8080/api/v1/orders/ORDER_ID/payments \
  -H "Authorization: Bearer ORGANIZER_TOKEN"

# 7) Change payment status (replace ORGANIZER_TOKEN and PAYMENT_ID)
curl -X PATCH http://localhost:8080/api/v1/payments/PAYMENT_ID/status \
  -H "Authorization: Bearer ORGANIZER_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"status":"CAPTURED"}'

# 8) Change order status (replace ORGANIZER_TOKEN and ORDER_ID)
curl -X PATCH http://localhost:8080/api/v1/orders/ORDER_ID/status \
  -H "Authorization: Bearer ORGANIZER_TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"status":"FUNDED"}'

# 9) Get user notifications (replace USER_TOKEN)
curl -X GET http://localhost:8080/api/v1/notifications/my \
  -H "Authorization: Bearer USER_TOKEN"

# 10) Mark item as distributed (replace ORGANIZER_TOKEN and ITEM_ID)
curl -X PATCH http://localhost:8080/api/v1/items/ITEM_ID/distributed \
  -H "Authorization: Bearer ORGANIZER_TOKEN" \
  -H "Content-Type: application/json"
